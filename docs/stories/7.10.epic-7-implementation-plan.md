```typescript
// Enhanced RetryHandler
async executeWithRetry<T>(operation: () => Promise<T>, context: RetryContext): Promise<T> {
  let lastError: Error;

  for (let attempt = 0; attempt <= context.maxRetries; attempt++) {
    try {
      return await operation();
    } catch (error) {
      lastError = error;

      // Special handling for authentication errors
      if (error.response?.status === 401 && attempt === 0) {
        // Refresh token or re-validate API key
        await this.authManager.validateCredentials();
      }

      // Don't retry authentication errors after first attempt
      if (error instanceof AuthenticationError) {
        throw error;
      }

      // Standard retry logic for other errors
      if (attempt < context.maxRetries && this.shouldRetry(error)) {
        await this.delay(calculateBackoff(attempt, context));
      }
    }
  }

  throw lastError;
}
```

## 3. 🟡 HIGH: Cross-Module Authentication Fixes

### 3.1 BaseClient Authentication Enhancement

**3.1.1 Centralized Authentication Logic**
```typescript
// Enhanced BaseClient
class BaseClient {
  private authManager: AuthManager;

  constructor(config: SDKConfig) {
    this.authManager = new AuthManager(config);
    this.setupInterceptors();
  }

  private setupInterceptors(): void {
    // Request interceptor - Add auth headers
    this.axios.interceptors.request.use(
      (config) => {
        config.headers = {
          ...config.headers,
          ...this.authManager.getAuthHeader()
        };
        return config;
      },
      (error) => Promise.reject(error)
    );

    // Response interceptor - Handle auth errors
    this.axios.interceptors.response.use(
      (response) => response,
      async (error) => {
        if (error.response?.status === 401) {
          // Try to refresh credentials
          try {
            await this.authManager.refreshCredentials();
            // Retry original request
            return this.axios.request(error.config);
          } catch (refreshError) {
            throw new AuthenticationError('Authentication failed. Please check your API key.');
          }
        }
        throw error;
      }
    );
  }
}
```

```typescript
// Enhanced date parameter support
interface TariffQueryOptions {
  date?: string | Date;
  dateFrom?: string | Date;
  dateTo?: string | Date;
  includeExpired?: boolean;
  includeFuture?: boolean;
}

class TariffsModule {
  async getTariffs(options?: TariffQueryOptions): Promise<TariffsResponse> {
    const params = new URLSearchParams();

    if (options?.date) {
      params.append('date', this.formatDate(options.date));
    }

    if (options?.dateFrom) {
      params.append('dateFrom', this.formatDate(options.dateFrom));
    }

    if (options?.dateTo) {
      params.append('dateTo', this.formatDate(options.dateTo));
    }

    if (options?.includeExpired) {
      params.append('includeExpired', 'true');
    }

    if (options?.includeFuture) {
      params.append('includeFuture', 'true');
    }

    const queryString = params.toString();
    const url = `/api/v1/tariffs${queryString ? `?${queryString}` : ''}`;

    return this.client.get(url);
  }

  private formatDate(date: string | Date): string {
    if (date instanceof Date) {
      return date.toISOString().split('T')[0]; // YYYY-MM-DD format
    }
    return date; // Assume already formatted string
  }
}
```

```typescript
class GeneralModule {
  /**
   * Get news and announcements for sellers
   * @param options - News query options
   * @returns News articles and announcements
   */
  async getNews(options?: NewsOptions): Promise<NewsResponse> {
    const params = new URLSearchParams();

    if (options?.page && options.page > 0) {
      params.append('page', options.page.toString());
    }

    if (options?.limit && options.limit > 0) {
      params.append('limit', Math.min(options.limit, 100).toString()); // Cap at 100
    }

    if (options?.category) {
      params.append('category', options.category);
    }

    if (options?.dateFrom) {
      params.append('dateFrom', this.formatDate(options.dateFrom));
    }

    if (options?.dateTo) {
      params.append('dateTo', this.formatDate(options.dateTo));
    }

    const queryString = params.toString();
    const url = `/api/v1/news${queryString ? `?${queryString}` : ''}`;

    return this.client.get(url);
  }

  private formatDate(date: string | Date): string {
    if (date instanceof Date) {
      return date.toISOString().split('T')[0];
    }
    return date;
  }
}
```

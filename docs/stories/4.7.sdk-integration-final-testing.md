  import { BaseClient } from './client/base-client';
  import { SDKConfig } from './config';
  import { GeneralModule } from './modules/general';
  import { ProductsModule } from './modules/products';
  import { OrdersFBSModule } from './modules/orders-fbs';
  import { OrdersFBWModule } from './modules/orders-fbw';
  import { FinancesModule } from './modules/finances';
  import { AnalyticsModule } from './modules/analytics';
  import { ReportsModule } from './modules/reports';
  import { CommunicationsModule } from './modules/communications';
  import { PromotionModule } from './modules/promotion';
  import { TariffsModule } from './modules/tariffs';
  import { InStorePickupModule } from './modules/in-store-pickup';

  /**
   * Wildberries API TypeScript SDK
   *
   * Provides type-safe access to all Wildberries marketplace API methods.
   * Includes automatic rate limiting, retry logic, and comprehensive error handling.
   *
   * @example
   * ```typescript
   * import { WildberriesSDK } from 'wb-api-sdk';
   *
   * const sdk = new WildberriesSDK({
   *   apiKey: process.env.WB_API_KEY
   * });
   *
   * // Use any module
   * const products = await sdk.products.getProductList();
   * const orders = await sdk.ordersFBS.getOrders();
   * const balance = await sdk.finances.getBalance();
   * ```
   *
   * @see {@link https://dev.wildberries.ru/} - Official API Documentation
   */
  export class WildberriesSDK {
    private readonly client: BaseClient;

    // All 11 API Modules
    public readonly general: GeneralModule;
    public readonly products: ProductsModule;
    public readonly ordersFBS: OrdersFBSModule;
    public readonly ordersFBW: OrdersFBWModule;
    public readonly finances: FinancesModule;
    public readonly analytics: AnalyticsModule;
    public readonly reports: ReportsModule;
    public readonly communications: CommunicationsModule;
    public readonly promotion: PromotionModule;
    public readonly tariffs: TariffsModule;
    public readonly inStorePickup: InStorePickupModule;

    /**
     * Create a new Wildberries SDK instance
     *
     * @param config - SDK configuration including API key and optional settings
     * @throws {ValidationError} If API key is missing or invalid
     */
    constructor(config: SDKConfig) {
      // Validate configuration
      if (!config.apiKey) {
        throw new ValidationError('API key is required', { field: 'apiKey' });
      }

      // Create shared BaseClient instance
      this.client = new BaseClient(config);

      // Initialize all modules with shared client
      this.general = new GeneralModule(this.client);
      this.products = new ProductsModule(this.client);
      this.ordersFBS = new OrdersFBSModule(this.client);
      this.ordersFBW = new OrdersFBWModule(this.client);
      this.finances = new FinancesModule(this.client);
      this.analytics = new AnalyticsModule(this.client);
      this.reports = new ReportsModule(this.client);
      this.communications = new CommunicationsModule(this.client);
      this.promotion = new PromotionModule(this.client);
      this.tariffs = new TariffsModule(this.client);
      this.inStorePickup = new InStorePickupModule(this.client);
    }

    /**
     * Test SDK connectivity and authentication
     *
     * @returns Promise<boolean> - True if connection successful
     */
    async ping(): Promise<boolean> {
      return this.general.ping();
    }
  }

  // Export SDK as default
  export default WildberriesSDK;

  // Re-export all types
  export * from './types';

  // Re-export all errors
  export * from './errors';

  // Re-export all modules for standalone use
  export { GeneralModule } from './modules/general';
  export { ProductsModule } from './modules/products';
  export { OrdersFBSModule } from './modules/orders-fbs';
  export { OrdersFBWModule } from './modules/orders-fbw';

  export { TariffsModule } from './modules/tariffs';

  // Re-export configuration interface
  export { SDKConfig } from './config';
  ```

    } catch (error) {

      }

    // Example 2: Rate Limit Error
    console.log('\n2. Handling Rate Limit Errors');
    try {
      // Make many requests quickly
      for (let i = 0; i < 100; i++) {
        await sdk.general.ping();
      }
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.error('   Rate limit exceeded!');
        console.error(`   Retry after: ${error.retryAfter}ms`);
        console.error('   Tip: Reduce request frequency or increase rate limits');
      }
    }

    // Example 3: Validation Error
    console.log('\n3. Handling Validation Errors');
    try {
      await sdk.products.createProduct({
        // Missing required fields
        name: ''
      } as any);
    } catch (error) {
      if (error instanceof ValidationError) {
        console.error('   Validation failed!');
        console.error(`   Field: ${error.details?.field}`);
        console.error(`   Error: ${error.message}`);
      }
    }

    // Example 4: Network Error
    console.log('\n4. Handling Network Errors');
    try {
      // Simulate network issue
      await sdk.products.getParentCategories();
    } catch (error) {
      if (error instanceof NetworkError) {
        console.error('   Network error occurred!');
        console.error('   Tip: Check internet connection');
        console.error(`   Error: ${error.message}`);
      }
    }

    // Example 5: Generic Error Handling
    console.log('\n5. Generic Error Handling');
    try {
      await sdk.products.getParentCategories();
    } catch (error) {
      if (error instanceof WBAPIError) {
        console.error(`   API Error: ${error.message}`);
        console.error(`   Status Code: ${error.statusCode}`);
        console.error(`   Response: ${JSON.stringify(error.response)}`);
      } else if (error instanceof Error) {
        console.error(`   Unexpected Error: ${error.message}`);
      } else {
        console.error('   Unknown error occurred');
      }
    }

    // Example 6: Retry with Error Handling
    console.log('\n6. Retry Pattern with Error Handling');
    async function retryOperation<T>(
      operation: () => Promise<T>,
      maxRetries: number = 3
    ): Promise<T> {
      let lastError: Error;

      for (let attempt = 1; attempt <= maxRetries; attempt++) {
        try {
          return await operation();
        } catch (error) {
          lastError = error as Error;

          if (error instanceof AuthenticationError) {
            // Don't retry auth errors
            throw error;
          }

          if (error instanceof RateLimitError) {
            // Wait for retry-after period
            console.log(`   Waiting ${error.retryAfter}ms before retry...`);
            await new Promise(resolve => setTimeout(resolve, error.retryAfter));
          } else if (attempt < maxRetries) {
            // Exponential backoff for other errors
            const delay = Math.pow(2, attempt) * 1000;
            console.log(`   Attempt ${attempt} failed, retrying in ${delay}ms...`);
            await new Promise(resolve => setTimeout(resolve, delay));
          }
        }
      }

      throw lastError!;
    }

    try {
      const result = await retryOperation(() => sdk.general.ping());
      console.log(`   Operation succeeded: ${result}`);
    } catch (error) {
      console.error(`   All retry attempts failed: ${(error as Error).message}`);
    }

    console.log('\n=== Error Handling Complete ===');
  }

### Key Integration Points

**Module Dependencies**:
All 11 modules depend on shared BaseClient instance:
```typescript
constructor(config: SDKConfig) {
  this.client = new BaseClient(config);

  // All modules share same client
  this.general = new GeneralModule(this.client);
  this.products = new ProductsModule(this.client);
  // ... etc
}
```

**Rate Limiter Coordination**:
Single RateLimiter instance (inside BaseClient) enforces limits across all modules using unique rate limit keys (e.g., `products.create`, `ordersFBS.getOrders`).

**Error Handling Consistency**:
All modules transform HTTP errors to typed WBAPIError hierarchy, ensuring consistent error handling experience.

**Type System Integration**:
Types exported from `src/types/index.ts` aggregate all module types for external usage.

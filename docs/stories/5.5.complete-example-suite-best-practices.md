```typescript
/**
 * Example: {Title}

 * - SDK installed: npm install wb-api-sdk
 * - {Additional prerequisites}
 *
 * Expected Output:
 * - {Output 1}
 * - {Output 2}
 * - {Output 3}
 *
 * Estimated Time: {X} minutes
 *
 * @see {@link https://docs.example.com/api/module} - API documentation
 */

import { WildberriesSDK, RateLimitError, AuthenticationError } from 'wb-api-sdk';

// Configuration
const API_KEY = process.env.WB_API_KEY;

if (!API_KEY) {
  console.error('❌ Error: WB_API_KEY environment variable not set');
  console.log('Set your API key: export WB_API_KEY="your_key_here"');
  process.exit(1);
}

// Initialize SDK
const sdk = new WildberriesSDK({
  apiKey: API_KEY
});

/**
 * Main function demonstrating {feature}
 */
async function main() {
  try {
    // Step 1: {Description}
    console.log('\n📍 Step 1: {Step description}...');
    const result1 = await sdk.module.method();
    console.log('✅ Success:', result1);
    // Expected output: {...}

    // Step 2: {Description}
    console.log('\n📍 Step 2: {Step description}...');
    const result2 = await sdk.module.method2();
    console.log('✅ Success:', result2);
    // Expected output: {...}

    // Final summary
    console.log('\n🎉 Example completed successfully!');

  } catch (error) {
    // Specific error handling
    if (error instanceof RateLimitError) {
      console.error('⚠️ Rate Limit Error:', error.message);
      console.log(`Retry after: ${error.retryAfter}ms`);
      // SDK automatically retries, but you can implement custom logic here
    } else if (error instanceof AuthenticationError) {
      console.error('🔐 Authentication Error:', error.message);
      console.log('Check your API key: echo $WB_API_KEY');
    } else {
      console.error('❌ Error:', error);
    }

    process.exit(1);
  }
}

// Run example
main();
```

### Error Handling Enhancement Pattern

**Standard Error Handling Template:**

```typescript
import {
  WildberriesSDK,
  WBAPIError,
  AuthenticationError,
  RateLimitError,
  ValidationError,
  NetworkError
} from 'wb-api-sdk';

async function exampleWithErrorHandling() {
  try {
    const result = await sdk.module.method();
    return result;

  } catch (error) {
    // Handle specific SDK errors
    if (error instanceof RateLimitError) {
      console.warn('⚠️ Rate limit exceeded');
      console.log(`Automatic retry in ${error.retryAfter}ms`);
      // SDK handles retry automatically
      // Optional: implement exponential backoff

    } else if (error instanceof AuthenticationError) {
      console.error('🔐 Authentication failed');
      console.log('Verify your API key is correct and active');
      throw error; // Don't retry auth errors

    } else if (error instanceof ValidationError) {
      console.error('❌ Validation error:', error.message);
      console.log('Check your input data:', error.details);
      throw error; // Fix input and retry

    } else if (error instanceof NetworkError) {
      console.error('🌐 Network error:', error.message);
      console.log('Check your internet connection');
      // Implement retry logic for network errors

    } else if (error instanceof WBAPIError) {
      console.error('⚠️ API error:', error.statusCode, error.message);
      // Handle other API errors

    } else {
      console.error('❌ Unexpected error:', error);
      throw error;
    }
  }
}
```

**Retry Logic Example:**

```typescript
async function withRetry<T>(
  operation: () => Promise<T>,
  maxRetries: number = 3
): Promise<T> {
  let lastError: Error;

  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await operation();
    } catch (error) {
      lastError = error;

      if (error instanceof AuthenticationError || error instanceof ValidationError) {
        throw error; // Don't retry these
      }

      if (attempt < maxRetries) {
        const delay = Math.pow(2, attempt) * 1000; // Exponential backoff
        console.log(`Retry ${attempt}/${maxRetries} after ${delay}ms...`);
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
  }

  throw lastError;
}

// Usage
const result = await withRetry(() => sdk.products.getParentCategories());
```

```markdown
- Verify Wildberries API status

**"Module Not Found"**
- Ensure SDK is installed: `npm install wb-api-sdk`
```

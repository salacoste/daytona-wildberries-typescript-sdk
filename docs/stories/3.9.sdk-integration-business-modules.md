    const MAX_SIZE_KB = 100;

    function validateBundleSize() {
      const bundlePath = 'dist/index.js';
      const content = readFileSync(bundlePath);
      const gzipped = gzipSync(content);
      const sizeKB = gzipped.length / 1024;

      console.log(`Bundle Size: ${sizeKB.toFixed(2)} KB (gzipped)`);
      console.log(`Target: <${MAX_SIZE_KB} KB`);

      if (sizeKB > MAX_SIZE_KB) {
        throw new Error(`Bundle size ${sizeKB.toFixed(2)}KB exceeds ${MAX_SIZE_KB}KB target`);
      }

      console.log('✅ Bundle size within target');
    }

    validateBundleSize();

### SDK Integration Pattern

Reference implementation from Epic 2 integration (Story 2.10):
```typescript
// src/index.ts
import { BaseClient } from './client/base-client';
import { GeneralModule } from './modules/general';
import { ProductsModule } from './modules/products';
import { OrdersFBSModule } from './modules/orders-fbs';
import type { SDKConfig } from './config';

export class WildberriesSDK {
  public readonly general: GeneralModule;
  public readonly products: ProductsModule;
  public readonly ordersFBS: OrdersFBSModule;

  constructor(config: SDKConfig) {
    const client = new BaseClient(config);

    this.general = new GeneralModule(client);
    this.products = new ProductsModule(client);
    this.ordersFBS = new OrdersFBSModule(client);
  }
}
```

Apply same pattern for Epic 3 modules - add four new properties and initialize with shared BaseClient.

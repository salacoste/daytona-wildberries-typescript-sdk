**Target Performance Benchmarks:**

| Metric | Target | Measurement |
|--------|--------|-------------|
| API Call Latency (p50) | <200ms | 50th percentile response time |
| API Call Latency (p95) | <500ms | 95th percentile response time |
| API Call Latency (p99) | <1000ms | 99th percentile response time |
| Throughput | >100 req/s | Requests per second |
| Memory Usage (idle) | <50MB | Resident set size |
| Memory Usage (load) | <200MB | Under 100 concurrent requests |
| Error Rate | <0.1% | Failed requests / total requests |

**Benchmarking Script Example:**

```typescript
// scripts/benchmark.ts
import { WildberriesSDK } from '../src';
import { performance } from 'perf_hooks';

const sdk = new WildberriesSDK({
  apiKey: process.env.WB_API_KEY!
});

async function benchmarkLatency(iterations = 100) {
  const latencies: number[] = [];

  for (let i = 0; i < iterations; i++) {
    const start = performance.now();
    await sdk.general.ping();
    const end = performance.now();
    latencies.push(end - start);
  }

  latencies.sort((a, b) => a - b);

  return {
    p50: latencies[Math.floor(iterations * 0.5)],
    p95: latencies[Math.floor(iterations * 0.95)],
    p99: latencies[Math.floor(iterations * 0.99)],
    avg: latencies.reduce((a, b) => a + b) / iterations
  };
}

async function benchmarkThroughput(duration = 10000) {
  const startTime = Date.now();
  let requests = 0;

  while (Date.now() - startTime < duration) {
    await sdk.general.ping();
    requests++;
  }

  const elapsed = (Date.now() - startTime) / 1000;
  return requests / elapsed;
}

async function main() {
  console.log('🚀 Running SDK Performance Benchmarks\n');

  console.log('📊 Latency Benchmark (100 requests)...');
  const latency = await benchmarkLatency();
  console.log(`  p50: ${latency.p50.toFixed(2)}ms`);
  console.log(`  p95: ${latency.p95.toFixed(2)}ms`);
  console.log(`  p99: ${latency.p99.toFixed(2)}ms`);
  console.log(`  avg: ${latency.avg.toFixed(2)}ms\n`);

  console.log('📊 Throughput Benchmark (10 seconds)...');
  const throughput = await benchmarkThroughput();
  console.log(`  Throughput: ${throughput.toFixed(2)} req/s\n`);

  console.log('✅ Benchmarks complete!');
}

main();
```

### Rate Limit Optimization Strategies

**Strategy 1: Parallel Requests to Different Endpoints**

```typescript
// Maximize throughput by calling different endpoints in parallel
async function fetchProductData(productId: string) {
  const [details, pricing, stock] = await Promise.all([
    sdk.products.getDetails(productId),    // Different rate limit
    sdk.products.getPricing(productId),    // Different rate limit
    sdk.products.getStock(productId)       // Different rate limit
  ]);

  return { details, pricing, stock };
}
```

**Strategy 2: Request Batching**

```typescript
// Batch multiple items into fewer API calls
async function updateProductsBatch(updates: ProductUpdate[]) {
  const batchSize = 50;  // API supports 50 items per request

  const batches = [];
  for (let i = 0; i < updates.length; i += batchSize) {
    batches.push(updates.slice(i, i + batchSize));
  }

  return Promise.all(
    batches.map(batch => sdk.products.updateBatch(batch))
  );
}
```

**Strategy 3: Intelligent Queueing**

```typescript
class RequestQueue {
  private queue: Array<() => Promise<any>> = [];
  private processing = false;
  private requestsPerSecond = 10;

  async enqueue<T>(fn: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push(async () => {
        try {
          const result = await fn();
          resolve(result);
        } catch (error) {
          reject(error);
        }
      });

      this.process();
    });
  }

  private async process() {
    if (this.processing) return;
    this.processing = true;

    while (this.queue.length > 0) {
      const fn = this.queue.shift()!;
      await fn();
      await this.sleep(1000 / this.requestsPerSecond);
    }

    this.processing = false;
  }

  private sleep(ms: number) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

### Caching Strategies

**In-Memory Caching Example:**

```typescript
import NodeCache from 'node-cache';

const cache = new NodeCache({
  stdTTL: 300,  // 5 minutes default TTL
  checkperiod: 60  // Check for expired keys every 60s
});

async function getCategories() {
  const cacheKey = 'categories:all';
  const cached = cache.get(cacheKey);

  if (cached) {
    return cached;
  }

  const categories = await sdk.products.getParentCategories();
  cache.set(cacheKey, categories);
  return categories;
}
```

**Redis Caching Example:**

```typescript
import Redis from 'ioredis';

const redis = new Redis({
  host: process.env.REDIS_HOST,
  port: 6379
});

async function getCategoriesWithRedis() {
  const cacheKey = 'wb:categories:all';
  const cached = await redis.get(cacheKey);

  if (cached) {
    return JSON.parse(cached);
  }

  const categories = await sdk.products.getParentCategories();
  await redis.setex(cacheKey, 300, JSON.stringify(categories));
  return categories;
}
```

**Cache Invalidation Strategy:**

```typescript
class SmartCache {
  private cache = new NodeCache();

  async get<T>(
    key: string,
    fetcher: () => Promise<T>,
    ttl: number = 300
  ): Promise<T> {
    const cached = this.cache.get<T>(key);
    if (cached) return cached;

    const data = await fetcher();
    this.cache.set(key, data, ttl);
    return data;
  }

  invalidate(pattern: string) {
    const keys = this.cache.keys();
    keys.filter(k => k.includes(pattern))
        .forEach(k => this.cache.del(k));
  }
}

// Usage
const smartCache = new SmartCache();

// Get with auto-cache
const categories = await smartCache.get(
  'categories',
  () => sdk.products.getParentCategories(),
  600  // 10 minutes
);

// Invalidate on update
await sdk.products.createCategory({...});
smartCache.invalidate('categories');
```

function badExample() {

  }, 1000);

// ✅ GOOD: Clear references when done
function goodExample() {
  const largeData = new Array(1000000).fill('data');
  const interval = setInterval(() => {
    console.log(largeData.length);

- [ ] **Task 4: Add estimated reading time** (AC: 4)
  - [ ] Create reading time plugin: .vitepress/theme/plugins/reading-time.ts:
    ```typescript
    import type { MarkdownRenderer } from 'vitepress';

    export function readingTimePlugin(md: MarkdownRenderer) {
      const render = md.render.bind(md);

      md.render = (src, env) => {
        const html = render(src, env);

        // Calculate reading time (average: 200 words per minute)
        const wordCount = src.split(/\s+/).length;
        const readingTime = Math.ceil(wordCount / 200);

        env.frontmatter = env.frontmatter || {};
        env.frontmatter.readingTime = readingTime;

        return html;
      };
    }
    ```

### Reading Time Estimation

**Reading time calculation:**

Average reading speed: 200-250 words per minute

```typescript
function calculateReadingTime(text: string): number {
  const wordsPerMinute = 200;
  const wordCount = text.split(/\s+/).length;
  return Math.ceil(wordCount / wordsPerMinute);
}
```

**Display criteria:**

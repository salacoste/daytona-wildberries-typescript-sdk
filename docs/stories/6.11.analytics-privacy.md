**VitePress integration pattern:**

```typescript
// .vitepress/config.ts
import { defineConfig } from 'vitepress';

const GA_MEASUREMENT_ID = process.env.GA_MEASUREMENT_ID || 'G-XXXXXXXXXX';

export default defineConfig({
  head: [
    // Google Analytics 4 - async loading (non-blocking)
    [
      'script',
      {
        async: true,
        src: `https://www.googletagmanager.com/gtag/js?id=${GA_MEASUREMENT_ID}`
      }
    ],
    [

      `

        gtag('config', '${GA_MEASUREMENT_ID}', {
          'anonymize_ip': true,              // Anonymize IP addresses (GDPR)
          'cookie_flags': 'SameSite=None;Secure',  // Secure cookies
          'cookie_expires': 63072000,        // 2 years
          'allow_google_signals': false,     // Disable Google signals
```

**Privacy-friendly settings explained:**
- `anonymize_ip: true` - Anonymizes last octet of IP (EU requirement)
- `cookie_flags` - Secure and SameSite attributes
- `allow_google_signals: false` - Disables cross-device tracking
- `allow_ad_personalization_signals: false` - No advertising data

### Custom Event Tracking Implementation

**Analytics helper module:**

```typescript
// .vitepress/theme/analytics.ts
declare global {
  interface Window {
    gtag: (...args: any[]) => void;
    dataLayer: any[];
  }
}

/**
 * Check if analytics is enabled (consent given)
 */
function isAnalyticsEnabled(): boolean {
  if (typeof window === 'undefined') return false;

  const consent = localStorage.getItem('cookie-consent');
  const optOut = localStorage.getItem('analytics-opt-out');

  return consent === 'accepted' && optOut !== 'true';
}

/**
 * Track page view (automatic on navigation)
 */
export function trackPageView(url: string) {
  if (!isAnalyticsEnabled() || typeof window.gtag === 'undefined') return;

  window.gtag('config', 'G-XXXXXXXXXX', {
    page_path: url,
    page_title: document.title
  });
}

/**
 * Track custom event
 */
export function trackEvent(eventName: string, eventParams?: Record<string, any>) {
  if (!isAnalyticsEnabled() || typeof window.gtag === 'undefined') return;

  window.gtag('event', eventName, eventParams);
}

/**
 * Track search query
 */
export function trackSearchQuery(query: string) {
  trackEvent('search', {
    search_term: query,
    page_location: window.location.href
  });
}

/**
 * Track conversion milestone
 */
export function trackConversion(milestone: string) {
  trackEvent('conversion', {
    milestone: milestone,
    timestamp: new Date().toISOString()
  });
}

/**
 * Track outbound link click
 */
export function trackOutboundLink(url: string) {
  trackEvent('click', {
    link_url: url,
    link_domain: new URL(url).hostname
  });
}
```

**Usage in VitePress theme:**

```typescript
// .vitepress/theme/index.ts
import DefaultTheme from 'vitepress/theme';
import { onMounted, watch } from 'vue';
import { useRoute } from 'vitepress';
import { trackPageView, trackOutboundLink } from './analytics';

export default {
  extends: DefaultTheme,
  setup() {
    const route = useRoute();

    // Track initial page view
    onMounted(() => {
      trackPageView(route.path);

      // Track outbound links
      document.addEventListener('click', (e) => {
        const target = e.target as HTMLElement;
        const link = target.closest('a');

        if (link && link.hostname !== window.location.hostname) {
          trackOutboundLink(link.href);
        }
      });
    });

    // Track client-side navigation
    watch(() => route.path, (newPath) => {
      trackPageView(newPath);
    });
  }
};
```

### Conversion Funnel Configuration

**Funnel stages:**
1. **Landing** - Homepage visit (`page_view` on `/`)
2. **Getting Started** - Quickstart page visit (`page_view` on `/getting-started/quickstart`)
3. **First API Call** - Code example viewed (`conversion` event: `viewed_quickstart_example`)

**Implementation:**

```typescript
// In quickstart.md or custom component
<script setup lang="ts">
import { onMounted } from 'vue';
import { trackConversion } from './.vitepress/theme/analytics';

onMounted(() => {
  // Track when user scrolls to first code example
  const codeBlock = document.querySelector('.vp-code-group');

  if (codeBlock) {
    const observer = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) {
        trackConversion('viewed_quickstart_example');
        observer.disconnect(); // Only track once
      }
    }, {
      threshold: 0.5 // 50% of code block visible
    });

    observer.observe(codeBlock);
  }
});
</script>
```

### Analytics Opt-Out Implementation

**Opt-out mechanism (GDPR Article 21):**

```html
<!-- docs/opt-out.md -->
<button @click="optOut">Disable Analytics Tracking</button>

<script setup>
function optOut() {
  // Method 1: GA4 disable property
  window['ga-disable-G-XXXXXXXXXX'] = true;

  // Method 2: LocalStorage flag
  localStorage.setItem('analytics-opt-out', 'true');

  // Method 3: Clear GA cookies
  document.cookie = '_ga=; Max-Age=0; path=/';
  document.cookie = '_ga_G-XXXXXXXXXX=; Max-Age=0; path=/';

  alert('Analytics tracking disabled. Your visits will no longer be tracked.');
}
</script>
```

**Persistent opt-out check:**

```typescript
// .vitepress/theme/index.ts
onMounted(() => {
```

### Common Issues and Solutions

**Issue 1: Analytics events not showing in GA4**
- Solution: Check GA4 real-time report (not standard reports - 24-48h delay)
- Verify Measurement ID correct
- Check browser console for gtag errors
- Test with GA4 DebugView

**Issue 2: Cookie consent banner not appearing**
- Solution: Clear localStorage
- Check if consent already given
- Verify component mounted correctly
- Inspect Vue devtools

**Issue 3: Analytics performance impact >100ms**
- Solution: Verify async loading attribute
- Check script placement (should be in <head>)
- Ensure no synchronous analytics calls
- Use defer instead of async if needed

**Issue 4: Opt-out not working**
- Solution: Check localStorage key
- Verify window['ga-disable-*'] syntax
- Clear cookies manually
- Test in incognito mode

**Issue 5: Search tracking not working**
- Solution: Algolia search integration needed (Story 6.5)
- Hook into search component
- Verify event name matches GA4 configuration
- Check search_term parameter

**Issue 6: Conversion funnel empty**
- Solution: Wait 24-48 hours for data processing
- Verify conversion event marked in GA4
- Check event parameters match
- Test with GA4 DebugView first

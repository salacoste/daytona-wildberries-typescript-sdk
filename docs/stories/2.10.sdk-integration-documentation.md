    import { ProductsModule } from './modules/products';
    import { OrdersFBSModule } from './modules/orders-fbs';
    import { OrdersFBWModule } from './modules/orders-fbw';
    import type { SDKConfig } from './config';

    /**
     * Main Wildberries SDK Class
     *
     * Aggregates all API modules with shared configuration.
     * Each module shares the same BaseClient instance for consistent
     * auth, rate limiting, and error handling.
     *
     * [Source: docs/architecture.md#Component-1-WildberriesSDK]
     */
    export class WildberriesSDK {
      /**
       * Products Module
       *
       * Category navigation, product CRUD, media/pricing management,
       * warehouse assignment, and stock management.
       *
       * [Source: Stories 2.1-2.4]
       */
      public readonly products: ProductsModule;

      /**
       * Orders FBS Module (Fulfillment by Seller)
       *
       * Order retrieval, status management, supply creation,
       * shipping label generation, and barcode/QR code retrieval.
       *
       * [Source: Stories 2.5-2.6]
       */
      public readonly ordersFBS: OrdersFBSModule;

      /**
       * Orders FBW Module (Fulfillment by Wildberries)
       *
       * Warehouse information, acceptance coefficients, supply tracking,
       * goods management, and transit tariffs.
       *
       * [Source: Story 2.7]
       */
      public readonly ordersFBW: OrdersFBWModule;

      /**
       * @param config - SDK configuration including API key and optional overrides
       */
      constructor(config: SDKConfig) {
        // Validate configuration
        if (!config.apiKey || config.apiKey.trim() === '') {
          throw new Error('API key is required. Provide via config.apiKey or WB_API_KEY environment variable.');
        }

        // Create shared BaseClient instance
        const client = new BaseClient(config);

        // Initialize modules with shared client
        this.products = new ProductsModule(client);
        this.ordersFBS = new OrdersFBSModule(client);
        this.ordersFBW = new OrdersFBWModule(client);
      }
    }

    // Export SDK class as default
    export default WildberriesSDK;

    # Rate Limit Configuration Guide

    Wildberries API enforces different rate limits per endpoint.
    The SDK automatically handles rate limiting using a token bucket algorithm.

    ## Rate Limit Tiers

    ### Products Module (3 tiers)

    **Tier 1: Category/Characteristic Operations (100 req/min)**
    - `getParentCategories`: 100 req/min, 600ms interval
    - `getCategories`: 100 req/min, 600ms interval
    - `getCharacteristics`: 100 req/min, 600ms interval

    **Tier 2: Product Creation/Media (1 req/10s)**
    - `createProduct`: 1 req/10s, 10s interval
    - `uploadMedia`: 1 req/10s, 10s interval
    - **CRITICAL**: Wait 10 seconds between operations

    **Tier 3: Pricing/Stock Updates (300-1000 req/min)**
    - `updatePricing`: 300 req/min, 200ms interval
    - `updateStock`: 1000 req/min, 60ms interval
    - **TIP**: Use bulk operations to maximize throughput

    ### Orders FBS Module (3 tiers)

    **Tier 1: Order Retrieval (100 req/min)**
    - `getOrders`: 100 req/min, 600ms interval
    - Use pagination for large result sets

    **Tier 2: Order Status/Supply (300 req/min)**
    - `updateOrderStatus`: 300 req/min, 200ms interval
    - `createSupply`: 300 req/min, 200ms interval
    - `deliverSupply`: 300 req/min, 200ms interval

    **Tier 3: Stickers/Barcodes (1000 req/min)**
    - `getOrderStickers`: 1000 req/min, 60ms interval
    - `getSupplyBarcode`: 1000 req/min, 60ms interval
    - **NOTE**: 409 errors count as 5 requests

    ### Orders FBW Module (2 tiers)

    **Tier 1: Planning Operations (6 req/min)**
    - `getWarehouses`: 6 req/min, 10s interval
    - `getAcceptanceCoefficients`: 6 req/min, 10s interval
    - `getAcceptanceOptions`: 6 req/min, 10s interval
    - `getTransitTariffs`: 6 req/min, 10s interval (burst: 10)
    - **CRITICAL**: Slowest tier, plan carefully

    **Tier 2: Supply Information (30 req/min)**
    - `getSupplies`: 30 req/min, 2s interval
    - `getSupplyDetails`: 30 req/min, 2s interval
    - `getSupplyGoods`: 30 req/min, 2s interval
    - `getSupplyPackage`: 30 req/min, 2s interval

    ## SDK Rate Limit Handling

    The SDK automatically enforces rate limits using a token bucket algorithm:

    ```typescript
    // Rate limits are enforced transparently
    for (let i = 0; i < 100; i++) {
      // SDK automatically spaces these requests at 600ms intervals
      await sdk.products.getParentCategories();
    }
    ```

    ## Custom Rate Limit Configuration

    Override default rate limits if needed:

    ```typescript
    const sdk = new WildberriesSDK({
      apiKey: process.env.WB_API_KEY,
      rateLimitConfig: {
        requestsPerSecond: 10,    // Global override
        requestsPerMinute: 600,   // Global override
        // Per-endpoint overrides
        'products.createProduct': {
          requestsPerMinute: 6,
          intervalSeconds: 10,
          burstLimit: 1
        }
      }
    });
    ```

    ## Best Practices

    1. **Cache When Possible**: Categories, characteristics, warehouses change infrequently
    2. **Batch Operations**: Use bulk updates for pricing/stock
    3. **Respect 10s Limits**: Product creation and media uploads are strictly limited
    4. **Plan FBW Queries**: Only 6 requests/min for warehouse planning
    5. **Monitor 409 Errors**: Count as 5 requests for sticker generation

      await sdk.products.createProduct(data);

      if (error instanceof RateLimitError) {
        console.log(`Wait ${error.retryAfter}ms before retrying`);
        await new Promise(r => setTimeout(r, error.retryAfter));
        // Retry operation
      }
    }

    ```typescript
    import { build } from 'vite';
    import { gzipSync } from 'zlib';
    import { statSync, readFileSync } from 'fs';

    async function validateBundleSize() {
      console.log('=== Bundle Size Validation ===\n');

      // Build full SDK
      await build({
        build: {
          lib: {
            entry: 'src/index.ts',
            formats: ['es']
          },
          rollupOptions: {
            external: ['axios']
          }
        }
      });

      const bundlePath = 'dist/index.es.js';
      const bundle = readFileSync(bundlePath);
      const gzipped = gzipSync(bundle);

      const rawSize = statSync(bundlePath).size;
      const gzipSize = gzipped.length;

      console.log(`Raw Size: ${(rawSize / 1024).toFixed(2)} KB`);
      console.log(`Gzipped Size: ${(gzipSize / 1024).toFixed(2)} KB`);

      const TARGET_SIZE = 100 * 1024; // 100KB
      const withinBudget = gzipSize <= TARGET_SIZE;

      if (withinBudget) {

      }
    }

    ```

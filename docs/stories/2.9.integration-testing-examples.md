    /**
     * Complete Product Lifecycle Workflow
     *
     * Demonstrates:
     * 1. Get parent categories
     * 2. Get category characteristics
     * 3. Create product card
     * 4. Upload product media
     * 5. Set pricing
     * 6. Assign to warehouse
     * 7. Update stock levels
     *
     * [Source: Stories 2.1-2.4]
     */
    import { WildberriesSDK } from '../src';
    import type { CreateProductRequest } from '../src/types/products.types';

    async function completeProductWorkflow() {
      const sdk = new WildberriesSDK({
        apiKey: process.env.WB_API_KEY!
      });

      console.log('=== Complete Product Workflow ===\n');

      try {
        // Step 1: Get parent categories
        console.log('1. Fetching parent categories...');
        const parents = await sdk.products.getParentCategories();
        const electronics = parents.data.find(p => p.name.includes('Электроника'));
        console.log(`✅ Found category: ${electronics.name} (ID: ${electronics.id})`);

        // Step 2: Get child categories
        console.log('\n2. Fetching child categories...');
        const children = await sdk.products.getCategories(electronics.id);
        const smartphones = children.data.find(c => c.name.includes('Смартфоны'));
        console.log(`✅ Found subcategory: ${smartphones.name} (ID: ${smartphones.id})`);

        // Step 3: Get characteristics
        console.log('\n3. Fetching required characteristics...');
        const chars = await sdk.products.getCharacteristics(smartphones.id);
        console.log(`✅ Retrieved ${chars.data.length} characteristics`);

        // Step 4: Create product
        console.log('\n4. Creating product card...');
        const productData: CreateProductRequest = {
          brandName: 'TechBrand',
          categoryId: smartphones.id,
          characteristics: [
            { id: 1, value: 'Android' },
            { id: 2, value: '128GB' }
          ],
          // ... other required fields from characteristics
        };
        const product = await sdk.products.createProduct(productData);
        console.log(`✅ Product created: ID ${product.id}`);

        // Step 5: Upload media (wait for rate limit)
        console.log('\n5. Uploading product images...');
        await new Promise(resolve => setTimeout(resolve, 10000)); // Rate limit: 1 req/10s

        const media = await sdk.products.uploadMedia(product.id, [
          { url: 'https://example.com/image1.jpg' },
          { url: 'https://example.com/image2.jpg' }
        ]);
        console.log(`✅ Uploaded ${media.data.length} images`);

        // Step 6: Set pricing
        console.log('\n6. Setting product pricing...');
        const pricing = await sdk.products.updatePricing([{
          productId: product.id,
          price: 29999,
          discount: 10
        }]);
        console.log(`✅ Pricing set: ${pricing.data[0].price}₽ (discount: ${pricing.data[0].discount}%)`);

        // Step 7: Assign warehouse
        console.log('\n7. Assigning to warehouse...');
        const warehouses = await sdk.products.getWarehouses();
        const warehouse = warehouses.data[0];
        await sdk.products.assignWarehouse(product.id, warehouse.id);
        console.log(`✅ Assigned to warehouse: ${warehouse.name}`);

        // Step 8: Update stock
        console.log('\n8. Updating stock levels...');
        const stock = await sdk.products.updateStock([{
          productId: product.id,
          warehouseId: warehouse.id,
          quantity: 100
        }]);
        console.log(`✅ Stock updated: ${stock.data[0].quantity} units`);

        console.log('\n=== Product Workflow Complete ===');
        console.log(`Product ID: ${product.id}`);
        console.log(`Status: Ready for sale`);

      } catch (error) {
        if (error instanceof RateLimitError) {
          console.error(`❌ Rate limit exceeded. Retry after ${error.retryAfter}ms`);
        } else if (error instanceof ValidationError) {
          console.error(`❌ Validation failed: ${error.message}`);
        } else {
          console.error(`❌ Error: ${error.message}`);
        }
        throw error;
      }
    }

    completeProductWorkflow().catch(console.error);

    /**
     * FBS Order Fulfillment Workflow
     *
     * Demonstrates:
     * 1. Fetch new orders
     * 2. Update order status
     * 3. Create supply
     * 4. Add orders to supply
     * 5. Get shipping stickers
     * 6. Deliver supply
     * 7. Get supply barcode
     *
     * [Source: Stories 2.5-2.6]
     */
    import { WildberriesSDK } from '../src';
    import { writeFileSync } from 'fs';

    async function fbsOrderFulfillment() {
      const sdk = new WildberriesSDK({
        apiKey: process.env.WB_API_KEY!
      });

      console.log('=== FBS Order Fulfillment Workflow ===\n');

      try {
        // Step 1: Fetch new orders
        console.log('1. Fetching new orders...');
        const orders = await sdk.ordersFBS.getOrders({
          status: 'new',
          dateFrom: new Date(Date.now() - 86400000).toISOString(), // Last 24h
          limit: 10
        });
        console.log(`✅ Found ${orders.orders.length} new orders`);

        if (orders.orders.length === 0) {
          console.log('No orders to process.');
          return;
        }

        const order = orders.orders[0];
        console.log(`Processing order: ${order.id}`);

        // Step 2: Update order status to confirmed
        console.log('\n2. Confirming order...');
        await sdk.ordersFBS.updateOrderStatus(order.id, 'confirm');
        console.log(`✅ Order ${order.id} confirmed`);

        // Step 3: Create supply
        console.log('\n3. Creating supply for shipment...');
        const supply = await sdk.ordersFBS.createSupply('Supply ' + new Date().toISOString());
        console.log(`✅ Supply created: ${supply.id}`);

        // Step 4: Add orders to supply
        console.log('\n4. Adding orders to supply...');
        await sdk.ordersFBS.addOrderToSupply(supply.id, order.id);
        console.log(`✅ Order ${order.id} added to supply ${supply.id}`);

        // Step 5: Get shipping stickers
        console.log('\n5. Generating shipping stickers...');
        const stickers = await sdk.ordersFBS.getOrderStickers(
          [order.id],
          { type: 'png', width: 58, height: 40 }
        );

        // Save sticker to file
        const stickerData = Buffer.from(stickers.stickers[0].file, 'base64');
        writeFileSync(`sticker-${order.id}.png`, stickerData);
        console.log(`✅ Sticker saved: sticker-${order.id}.png`);

        // Step 6: Mark supply as delivered
        console.log('\n6. Marking supply as delivered...');
        await sdk.ordersFBS.deliverSupply(supply.id);
        console.log(`✅ Supply ${supply.id} delivered`);

        // Step 7: Get supply barcode
        console.log('\n7. Getting supply QR code...');
        const barcode = await sdk.ordersFBS.getSupplyBarcode(supply.id, 'png');
        const barcodeData = Buffer.from(barcode.file, 'base64');
        writeFileSync(`supply-qr-${supply.id}.png`, barcodeData);
        console.log(`✅ QR code saved: supply-qr-${supply.id}.png`);

        console.log('\n=== FBS Fulfillment Complete ===');
        console.log(`Orders processed: ${orders.orders.length}`);
        console.log(`Supply ID: ${supply.id}`);
        console.log(`Files created: sticker-${order.id}.png, supply-qr-${supply.id}.png`);

      } catch (error) {
        if (error instanceof RateLimitError) {
          console.error(`❌ Rate limit exceeded. Retry after ${error.retryAfter}ms`);
        } else if (error instanceof ValidationError) {
          console.error(`❌ Validation failed: ${error.message}`);
        } else {
          console.error(`❌ Error: ${error.message}`);
        }
        throw error;
      }
    }

    fbsOrderFulfillment().catch(console.error);

    /**
     * FBW Supply Management Workflow
     *
     * Demonstrates:
     * 1. Get available warehouses
     * 2. Check acceptance coefficients
     * 3. Verify acceptance options for goods
     * 4. Get existing supplies
     * 5. Get supply details
     * 6. Retrieve goods in supply
     * 7. Get package information
     * 8. Check transit tariffs
     *
     * [Source: Story 2.7]
     */
    import { WildberriesSDK } from '../src';

    async function fbwSupplyManagement() {
      const sdk = new WildberriesSDK({
        apiKey: process.env.WB_API_KEY!
      });

      console.log('=== FBW Supply Management Workflow ===\n');

      try {
        // Step 1: Get available warehouses
        console.log('1. Fetching WB warehouses...');
        const warehouses = await sdk.ordersFBW.getWarehouses();
        const active = warehouses.filter(w => w.isActive);
        console.log(`✅ Found ${active.length} active warehouses`);
        const warehouse = active[0];
        console.log(`   Using: ${warehouse.name} (${warehouse.address})`);

        // Step 2: Check acceptance coefficients
        console.log('\n2. Checking acceptance coefficients...');
        const coeffs = await sdk.ordersFBW.getAcceptanceCoefficients(warehouse.ID.toString());
        const available = coeffs.filter(c =>
          (c.coefficient === 0 || c.coefficient === 1) && c.allowUnload
        );
        console.log(`✅ ${available.length} dates with available acceptance`);
        if (available.length > 0) {
          console.log(`   Next available: ${available[0].date} (coeff: ${available[0].coefficient})`);
        }

        // Step 3: Check acceptance options for goods
        console.log('\n3. Checking acceptance options for products...');
        const goods = [
          { barcode: '1234567891234', quantity: 10 },
          { barcode: '9876543210987', quantity: 5 }
        ];
        const options = await sdk.ordersFBW.getAcceptanceOptions(goods);
        options.result.forEach(opt => {
          if (opt.isError) {
            console.log(`   ❌ ${opt.barcode}: ${opt.error.detail}`);
          } else {
            console.log(`   ✅ ${opt.barcode}: ${opt.warehouses.length} warehouses`);
          }
        });

        // Step 4: Get recent supplies
        console.log('\n4. Fetching recent supplies...');
        const supplies = await sdk.ordersFBW.getSupplies({
          dates: [{
            from: '2024-01-01',
            till: '2024-12-31',
            type: 'createDate'
          }],
          statusIDs: [2, 3, 4, 5] // Planned, AllowUnload, Accepting, Accepted
        }, 10);
        console.log(`✅ Found ${supplies.length} supplies`);

        if (supplies.length > 0) {
          const supply = supplies[0];
          console.log(`   Latest: ${supply.supplyID} (${supply.statusName})`);

          // Step 5: Get supply details
          console.log('\n5. Getting supply details...');
          const details = await sdk.ordersFBW.getSupplyDetails(supply.supplyID!);
          console.log(`✅ Supply details retrieved`);
          console.log(`   Warehouse: ${details.warehouseName}`);
          console.log(`   Box type: ${details.boxTypeName}`);
          console.log(`   Quantities: added=${details.quantity}, accepted=${details.acceptedQuantity}`);

          // Step 6: Get goods in supply
          console.log('\n6. Getting goods in supply...');
          const supplyGoods = await sdk.ordersFBW.getSupplyGoods(supply.supplyID!);
          console.log(`✅ Found ${supplyGoods.length} products`);
          supplyGoods.slice(0, 3).forEach(good => {
            console.log(`   - ${good.vendorCode}: ${good.quantity} units`);
          });

          // Step 7: Get package info
          console.log('\n7. Getting package information...');
          const boxes = await sdk.ordersFBW.getSupplyPackage(supply.supplyID!);
          console.log(`✅ Found ${boxes.length} packages`);
        }

        // Step 8: Check transit tariffs
        console.log('\n8. Checking transit tariffs...');
        const tariffs = await sdk.ordersFBW.getTransitTariffs();
        console.log(`✅ Found ${tariffs.length} transit routes`);
        if (tariffs.length > 0) {
          const tariff = tariffs[0];
          console.log(`   ${tariff.transitWarehouseName} → ${tariff.destinationWarehouseName}`);
          if (tariff.palletTariff) {
            console.log(`   Pallet: ${tariff.palletTariff}₽`);
          }
        }

        console.log('\n=== FBW Workflow Complete ===');

      } catch (error) {
        if (error instanceof RateLimitError) {
          console.error(`❌ Rate limit exceeded. Retry after ${error.retryAfter}ms`);
        } else if (error instanceof ValidationError) {
          console.error(`❌ Validation failed: ${error.message}`);
        } else {
          console.error(`❌ Error: ${error.message}`);
        }
        throw error;
      }
    }

    fbwSupplyManagement().catch(console.error);

    import {

      console.log('1. Handling Rate Limit Errors...');
      try {
        for (let i = 0; i < 10; i++) {
          await sdk.products.createProduct({ ... });
        }
      } catch (error) {
        if (error instanceof RateLimitError) {
          console.log(`⚠️  Rate limit hit. Retrying after ${error.retryAfter}ms`);
          await new Promise(r => setTimeout(r, error.retryAfter));
          // Retry the operation
        }
      }

      // Example 2: Validation Error
      console.log('\n2. Handling Validation Errors...');
      try {
        await sdk.products.createProduct({
          brandName: '', // Invalid: empty
          categoryId: -1 // Invalid: negative
        } as any);
      } catch (error) {
        if (error instanceof ValidationError) {
          console.log(`❌ Validation failed: ${error.message}`);
          console.log(`   Fields: ${JSON.stringify(error.fields)}`);
        }
      }

      // Example 3: Authentication Error
      console.log('\n3. Handling Authentication Errors...');
      const invalidSdk = new WildberriesSDK({ apiKey: 'invalid' });
      try {
        await invalidSdk.products.getParentCategories();
      } catch (error) {
        if (error instanceof AuthenticationError) {
          console.log(`❌ Auth failed: ${error.message}`);
          console.log('   Please check your API key');
        }
      }

      // Example 4: Network Error with Retry
      console.log('\n4. Handling Network Errors with Auto-Retry...');
      try {
        const result = await sdk.ordersFBS.getOrders({ limit: 10 });
        console.log(`✅ Retrieved ${result.orders.length} orders (with auto-retry)`);
      } catch (error) {
        if (error instanceof NetworkError) {
          console.log(`❌ Network error after all retries: ${error.message}`);
        }
      }

      console.log('\n=== Error Handling Complete ===');

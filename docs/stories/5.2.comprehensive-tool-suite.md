### Tool Generation Strategy

**Approach:** Automated code generation from SDK TypeScript definitions

**Tool Generator Implementation Pattern:**
```typescript
// mcp-server/tools/generate-mcp-tools.ts
import * as ts from 'typescript';
import { Project } from 'ts-morph';

interface ToolDefinition {
  name: string;            // e.g., "products_create"
  description: string;     // From JSDoc
  module: string;          // e.g., "products"
  method: string;          // e.g., "createProduct"
  inputSchema: object;     // JSON Schema from TypeScript types
  examples: string[];      // From JSDoc examples
}

function generateToolsFromSDK(): ToolDefinition[] {
  const project = new Project({
    tsConfigFilePath: '../tsconfig.json'
  });

  const sdkFile = project.getSourceFile('../src/index.ts');
  const tools: ToolDefinition[] = [];

  // Find WildberriesSDK class
  const sdkClass = sdkFile.getClass('WildberriesSDK');

  // Iterate over module properties (general, products, etc.)
  sdkClass.getProperties().forEach(prop => {
    const moduleName = prop.getName();
    const moduleType = prop.getType();

    // Get module class methods
    const methods = moduleType.getProperties();

    methods.forEach(method => {
      const tool = generateToolFromMethod(moduleName, method);
      tools.push(tool);
    });
  });

  return tools;
}

function generateToolFromMethod(module: string, method: Symbol): ToolDefinition {
  const methodName = method.getName();
  const signature = method.getDeclarations()[0] as ts.MethodDeclaration;

  // Extract JSDoc
  const jsDoc = signature.getJsDocs()[0];
  const description = jsDoc?.getDescription() || '';

  // Generate JSON Schema from TypeScript parameters
  const inputSchema = generateSchemaFromParameters(signature.getParameters());

  return {
    name: `${module}_${camelToSnake(methodName)}`,
    description,
    module,
    method: methodName,
    inputSchema,
    examples: extractExamples(jsDoc)
  };
}
```

**Conversion Function:**
```typescript
function methodNameToToolName(module: string, methodName: string): string {
  // createProduct → products_create
  // getOrderById → orders_fbs_get
  // updateOrderStatus → orders_fbs_update_status

  const snake = methodName
    .replace(/([A-Z])/g, '_$1')
    .toLowerCase()
    .replace(/^_/, '');

  return `${module}_${snake}`;
}
```

// mcp-server/src/mappers/parameter-mapper.ts
export class ParameterMapper {
  mapToSDKType<T>(mcpArgs: any, targetType: string): T {
    // Handle nested objects
    if (targetType === 'ProductFilters') {
      return {
        limit: mcpArgs.limit,
        offset: mcpArgs.offset,
        status: this.mapEnum(mcpArgs.status, ProductStatus),
        categoryId: mcpArgs.categoryId,
        dateRange: mcpArgs.dateRange ? {
          from: this.normalizeDate(mcpArgs.dateRange.from),
          to: this.normalizeDate(mcpArgs.dateRange.to)
        } : undefined
      } as T;
    }
    // ... handle other types
  }

  private mapEnum(value: string, enumType: any): any {
    if (!value) return undefined;

    // Validate enum value
    const validValues = Object.values(enumType);
    if (!validValues.includes(value)) {
      throw new ValidationError(`Invalid enum value: ${value}. Must be one of: ${validValues.join(', ')}`);
    }

    return value;
  }

  private normalizeDate(dateStr: string): string {
    // Ensure ISO 8601 format
    const date = new Date(dateStr);
    if (isNaN(date.getTime())) {
      throw new ValidationError(`Invalid date format: ${dateStr}. Use ISO 8601 format.`);
    }
    return date.toISOString();
  }
}

```typescript
// mcp-server/src/resources/documentation.ts
import { readFileSync } from 'fs';
import { join } from 'path';

export class DocumentationResourceHandler {
  private sdkExamplesPath = '../examples';

  async handleResourceRead(uri: string): Promise<{ contents: Array<{ uri: string; mimeType?: string; text?: string }> }> {
    // Parse URI: wildberries://docs/{module}
    const match = uri.match(/^wildberries:\/\/docs\/(.+)$/);
    if (!match) {
      throw new Error(`Invalid resource URI: ${uri}`);
    }

    const module = match[1];
    const content = this.generateModuleDocumentation(module);

    return {
      contents: [{
        uri,
        mimeType: 'text/markdown',
        text: content
      }]
    };
  }

  private generateModuleDocumentation(module: string): string {
    // Extract from TypeDoc JSON
    const typeDocData = this.loadTypeDocData(module);

    // Load examples
    const examples = this.loadExamples(module);

    // Generate markdown
    return `
# ${module.charAt(0).toUpperCase() + module.slice(1)} Module

${typeDocData.description}

## Available Operations

${typeDocData.methods.map(m => `- **${m.name}**: ${m.description}`).join('\n')}

## Examples

${examples}

## Rate Limits

${this.extractRateLimits(module)}
    `.trim();
  }

  private loadExamples(module: string): string {
    const exampleFile = join(this.sdkExamplesPath, `${module}.ts`);
    try {
      return '```typescript\n' + readFileSync(exampleFile, 'utf-8') + '\n```';
    } catch {
      return 'No examples available for this module.';
    }
  }
}
```

import { readFileSync } from 'fs';
import Ajv from 'ajv';

export function loadConfiguration(): MCPServerConfig {
  const configPath = process.env.WB_MCP_CONFIG_PATH || './mcp-config.json';

  let config: MCPServerConfig;
  try {
    const raw = readFileSync(configPath, 'utf-8');
    config = JSON.parse(raw);
  } catch (error) {
    console.warn(`No config file found at ${configPath}, using defaults`);
    return getDefaultConfig();
  }

  // Validate against schema
  const ajv = new Ajv();
  const validate = ajv.compile(configSchema);

  if (!validate(config)) {
    throw new Error(`Invalid configuration: ${JSON.stringify(validate.errors)}`);
  }

  return config;
}

### Tool Filtering Logic

```typescript
// mcp-server/src/tools/filter.ts
export class ToolFilter {
  constructor(private config: MCPServerConfig) {}

  filterTools(allTools: Tool[]): Tool[] {
    const mode = this.config.tools?.mode || 'all';

    if (mode === 'all') {
      return allTools;
    }

    if (mode === 'allowlist') {
      return this.applyAllowlist(allTools);
    }

    if (mode === 'blocklist') {
      return this.applyBlocklist(allTools);
    }

    return allTools;
  }

  private applyAllowlist(allTools: Tool[]): Tool[] {
    const { modules, operations, explicit } = this.config.tools;

    return allTools.filter(tool => {
      // Explicit tool names take precedence
      if (explicit?.includes(tool.name)) {
        return true;
      }

      // Filter by module
      const [module, operation] = tool.name.split('_');
      if (modules && !modules.includes(module)) {
        return false;
      }

      // Filter by operation type
      if (operations && !this.matchesOperation(operation, operations)) {
        return false;
      }

      return true;
    });
  }

  private matchesOperation(operation: string, allowedOps: string[]): boolean {
    return allowedOps.some(op => operation.includes(op));
  }
}
```

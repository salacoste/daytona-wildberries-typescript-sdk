## Story

**As a** documentation maintainer,
**I want** automated validation of documentation quality,
**so that** broken links, invalid code examples, and formatting issues are caught before deployment.

## Acceptance Criteria

1. Link validation script integrated into CI/CD (npm run validate:links)
2. Internal link checker validates all markdown links resolve correctly
3. External link checker validates external URLs return 200 status (with timeout handling)
4. Code example validation ensures all TypeScript snippets compile (npm run validate:examples)
5. Markdown linter validates frontmatter metadata (title, description required)
6. Image reference validation ensures all image paths resolve
7. Validation runs in GitHub Actions before deployment
8. Validation failures block deployment to GitHub Pages
9. Validation report generated listing all issues found
10. Manual override available for false positives (e.g., external links temporarily down)

    ```javascript
    async function checkExternalLink(url, retries = 2) {
      for (let i = 0; i <= retries; i++) {
        try {
          const response = await axios.head(url, {
            timeout: 5000,
            maxRedirects: 5,
            validateStatus: (status) => status < 500 // Accept 2xx, 3xx, 4xx
          });

          if (response.status >= 200 && response.status < 400) {
            return { status: 'alive', statusCode: response.status };
          }
        } catch (error) {
          if (i === retries) {
            return { status: 'dead', statusCode: error.code, error: error.message };
          }
          await sleep(1000 * (i + 1)); // Exponential backoff
        }
      }
    }
    ```

    ```javascript
    import fs from 'fs';
    import path from 'path';
    import { glob } from 'glob';

    const docsDir = path.resolve('docs');
    const publicDir = path.resolve('docs/public');
    const markdownFiles = glob.sync('**/*.md', { cwd: docsDir });

    let totalImages = 0;
    let missingImages = 0;

    for (const file of markdownFiles) {
      const filePath = path.join(docsDir, file);
      const content = fs.readFileSync(filePath, 'utf8');

      // Extract image references: ![alt](path)
      const imageRegex = /!\[.*?\]\((.*?)\)/g;
      let match;

      while ((match = imageRegex.exec(content)) !== null) {

          resolvedPath = path.join(path.dirname(filePath), imagePath);
        }

        if (!fs.existsSync(resolvedPath)) {
          console.error(`❌ ${file}: Missing image ${imagePath}`);
          missingImages++;
        }
      }
    }

    if (missingImages > 0) {
      console.error(`❌ Found ${missingImages} missing images out of ${totalImages}`);
      process.exit(1);
    ```

### Enhanced Link Validation Implementation

**Comprehensive link checker:**

```javascript
// scripts/validate-links.js
import fs from 'fs';
import path from 'path';
import { glob } from 'glob';
import axios from 'axios';
import { parse } from 'node-html-parser';

// Load configuration
const config = JSON.parse(fs.readFileSync('.validation-config.json', 'utf8'));

const distDir = path.resolve('.vitepress/dist');
const htmlFiles = glob.sync('**/*.html', { cwd: distDir });

let totalLinks = 0;
let brokenLinks = [];

console.log('🔍 Validating documentation links...\n');

for (const file of htmlFiles) {
  const filePath = path.join(distDir, file);
  const content = fs.readFileSync(filePath, 'utf8');
  const root = parse(content);

  const links = root.querySelectorAll('a[href]');

  for (const link of links) {
    const href = link.getAttribute('href');
    totalLinks++;

    // Skip ignored patterns
    if (shouldIgnore(href, config.ignorePatterns)) {
      continue;
    }

    // Validate internal links
    if (isInternalLink(href)) {
      const result = validateInternalLink(href, distDir);
      if (!result.valid) {
        brokenLinks.push({
          file,
          link: href,
          type: 'internal',
          error: result.error
        });
      }
    }
    // Validate external links
    else if (isExternalLink(href)) {
      const result = await validateExternalLink(href, config);
      if (!result.valid) {
        brokenLinks.push({
          file,
          link: href,
          type: 'external',
          error: result.error,
          statusCode: result.statusCode
        });
      }
    }
  }
}

// Report results
console.log(`\n📊 Validation Results:`);
console.log(`   Total links checked: ${totalLinks}`);
console.log(`   Broken links: ${brokenLinks.length}\n`);

if (brokenLinks.length > 0) {
  console.error('❌ Broken links found:\n');
  brokenLinks.forEach(link => {
    console.error(`   ${link.file}`);
    console.error(`   └─ ${link.link} (${link.type}) - ${link.error}`);
  });

  if (process.argv.includes('--ignore-external')) {
    console.log('\n⚠️  External link failures ignored (--ignore-external flag)');
    const internalBroken = brokenLinks.filter(l => l.type === 'internal');
    if (internalBroken.length > 0) {
      process.exit(1);
    }
  } else {
    process.exit(1);
  }
} else {
  console.log('✅ All links are valid!');
}

function isInternalLink(href) {
  return href.startsWith('/') || href.startsWith('./') || href.startsWith('../') || href.startsWith('#');
}

function isExternalLink(href) {
  return href.startsWith('http://') || href.startsWith('https://');
}

function shouldIgnore(href, patterns) {
  return patterns.externalLinks.some(pattern => href.includes(pattern));
}

function validateInternalLink(href, baseDir) {
  // Remove hash fragment
  const pathWithoutHash = href.split('#')[0];

  if (!pathWithoutHash) {
    return { valid: true }; // Just a hash link
  }

  // Resolve to file path
  let resolvedPath = path.join(baseDir, pathWithoutHash);

  // Check if file exists
  if (fs.existsSync(resolvedPath)) {
    return { valid: true };
  }

  // Try with .html extension
  if (fs.existsSync(resolvedPath + '.html')) {
    return { valid: true };
  }

  // Try as directory with index.html
  if (fs.existsSync(path.join(resolvedPath, 'index.html'))) {
    return { valid: true };
  }

  return {
    valid: false,
    error: 'File not found'
  };
}

async function validateExternalLink(url, config) {
  for (let attempt = 0; attempt <= config.externalLinkRetries; attempt++) {
    try {
      const response = await axios.head(url, {
        timeout: config.externalLinkTimeout,
        maxRedirects: 5,
        validateStatus: (status) => status < 500
      });

      if (response.status >= 200 && response.status < 400) {
        return { valid: true, statusCode: response.status };
      } else {
        return {
          valid: false,
          statusCode: response.status,
          error: `HTTP ${response.status}`
        };
      }
    } catch (error) {
      if (attempt === config.externalLinkRetries) {
        return {
          valid: false,
          error: error.code || error.message,
          statusCode: error.response?.status
        };
      }
      await new Promise(resolve => setTimeout(resolve, 1000 * (attempt + 1)));
    }
  }
}
```

// scripts/validate-doc-examples.ts
import { glob } from 'glob';
import fs from 'fs';
import path from 'path';
import ts from 'typescript';

const docsDir = path.resolve('docs');
const markdownFiles = glob.sync('**/*.md', { cwd: docsDir });

let totalExamples = 0;
let invalidExamples: Array<{
  file: string;
  line: number;
  code: string;
  errors: string[];
}> = [];

console.log('🔍 Validating TypeScript code examples...\n');

for (const file of markdownFiles) {
  const filePath = path.join(docsDir, file);
  const content = fs.readFileSync(filePath, 'utf8');

  // Extract TypeScript code blocks
  const codeBlocks = extractCodeBlocks(content, ['typescript', 'ts']);

  for (const codeBlock of codeBlocks) {
    totalExamples++;

    // Skip if marked with @ts-nocheck
    if (codeBlock.code.includes('// @ts-nocheck') ||
        codeBlock.code.includes('// @ts-ignore-file')) {
      continue;
    }

    const diagnostics = validateTypeScriptCode(codeBlock.code);

    if (diagnostics.length > 0) {
      invalidExamples.push({
        file,
        line: codeBlock.line,
        code: codeBlock.code.substring(0, 100) + '...',
        errors: diagnostics.map(d =>
          ts.flattenDiagnosticMessageText(d.messageText, '\n')
        )
      });
    }
  }
}

console.log(`📊 Validation Results:`);
console.log(`   Total code examples: ${totalExamples}`);

    example.errors.forEach(err => {
      console.error(`   └─ ${err}`);

function extractCodeBlocks(markdown: string, languages: string[]) {
  const blocks: Array<{ code: string; line: number; language: string }> = [];
  const lines = markdown.split('\n');

  let inCodeBlock = false;
  let currentBlock: string[] = [];
  let blockLanguage = '';
  let blockStartLine = 0;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];

    if (line.startsWith('```')) {
      if (!inCodeBlock) {
        // Start of code block
        const lang = line.substring(3).trim();
        if (languages.includes(lang)) {
          inCodeBlock = true;
          blockLanguage = lang;
          blockStartLine = i + 1;
          currentBlock = [];
        }
      } else {
        // End of code block
        blocks.push({
          code: currentBlock.join('\n'),
          line: blockStartLine,
          language: blockLanguage
        });
        inCodeBlock = false;
      }
    } else if (inCodeBlock) {
      currentBlock.push(line);
    }
  }

  return blocks;
}

function validateTypeScriptCode(code: string): ts.Diagnostic[] {
  const options: ts.CompilerOptions = {
    target: ts.ScriptTarget.ES2020,
    module: ts.ModuleKind.ESNext,
    strict: true,
    skipLibCheck: true,
    noEmit: true,
    types: ['node']
  };

  // Create virtual source file
  const sourceFile = ts.createSourceFile(
    'example.ts',
    code,
    ts.ScriptTarget.ES2020,
    true
  );

  // Create program
  const host = ts.createCompilerHost(options);
  const originalGetSourceFile = host.getSourceFile;

  host.getSourceFile = (fileName, languageVersion) => {
    if (fileName === 'example.ts') {
      return sourceFile;
    }
    return originalGetSourceFile(fileName, languageVersion);
  };

  const program = ts.createProgram(['example.ts'], options, host);

  // Get diagnostics
  const diagnostics = ts.getPreEmitDiagnostics(program);

  // Filter out import-related errors (common in examples)
  return diagnostics.filter(d => {
    const messageText = ts.flattenDiagnosticMessageText(d.messageText, '\n');
    return !messageText.includes('Cannot find module') &&
           !messageText.includes('Cannot find name');
  });
}

```javascript
// scripts/validate-frontmatter.js
import fs from 'fs';
import path from 'path';
import { glob } from 'glob';
import yaml from 'js-yaml';

const docsDir = path.resolve('docs');
const markdownFiles = glob.sync('**/*.md', { cwd: docsDir });

let totalFiles = 0;
let missingFrontmatter = [];

console.log('🔍 Validating frontmatter metadata...\n');

for (const file of markdownFiles) {
  // Skip certain files
  if (file.includes('node_modules') || file.includes('drafts')) {
    continue;
  }

  totalFiles++;

  const filePath = path.join(docsDir, file);
  const content = fs.readFileSync(filePath, 'utf8');

  const frontmatter = extractFrontmatter(content);

  if (!frontmatter) {
    missingFrontmatter.push({
      file,
      error: 'No frontmatter found'
    });
    continue;
  }

  // Validate required fields
  if (!frontmatter.title) {
    missingFrontmatter.push({
      file,
      error: "Missing 'title' field"
    });
  }

  if (!frontmatter.description) {
    missingFrontmatter.push({
      file,
      error: "Missing 'description' field"
    });
  }

  // Validate description length (150-160 characters for SEO)
  if (frontmatter.description && frontmatter.description.length > 160) {
    missingFrontmatter.push({
      file,
      error: `Description too long: ${frontmatter.description.length} chars (max 160)`
    });
  }
}

console.log(`📊 Validation Results:`);
console.log(`   Total files checked: ${totalFiles}`);
console.log(`   Files with issues: ${missingFrontmatter.length}\n`);

if (missingFrontmatter.length > 0) {
  console.error('❌ Frontmatter validation failed:\n');
  missingFrontmatter.forEach(issue => {
    console.error(`   ${issue.file}`);
    console.error(`   └─ ${issue.error}\n`);
  });
  process.exit(1);
} else {
  console.log('✅ All frontmatter is valid!');
}

function extractFrontmatter(markdown: string) {
  const match = markdown.match(/^---\n([\s\S]*?)\n---/);
  if (!match) return null;

  try {
    return yaml.load(match[1]);
  } catch (error) {
    return null;
  }
}
```

### Image Reference Validation

**Image path validator:**

```javascript
// scripts/validate-image-references.js
import fs from 'fs';
import path from 'path';
import { glob } from 'glob';

const docsDir = path.resolve('docs');
const publicDir = path.resolve('docs/public');
const markdownFiles = glob.sync('**/*.md', { cwd: docsDir });

let totalImages = 0;
let missingImages = [];
let missingAltText = [];

console.log('🔍 Validating image references...\n');

for (const file of markdownFiles) {
  const filePath = path.join(docsDir, file);
  const content = fs.readFileSync(filePath, 'utf8');

  // Extract image references: ![alt](path)
  const imageRegex = /!\[(.*?)\]\((.*?)\)/g;
  let match;

  while ((match = imageRegex.exec(content)) !== null) {
    totalImages++;

    const altText = match[1];
    const imagePath = match[2];

    // Check alt text
    if (!altText || altText.trim() === '') {
      missingAltText.push({
        file,
        image: imagePath
      });
    }

    // Skip external images
    if (imagePath.startsWith('http://') || imagePath.startsWith('https://')) {
      continue;
    }

    // Resolve image path
    let resolvedPath;
    if (imagePath.startsWith('/')) {
      // Absolute path from public directory
      resolvedPath = path.join(publicDir, imagePath);
    } else {
      // Relative path from markdown file
      resolvedPath = path.join(path.dirname(filePath), imagePath);
    }

    // Check if image exists
    if (!fs.existsSync(resolvedPath)) {
      missingImages.push({
        file,
        image: imagePath,
        resolved: resolvedPath
      });
    }
  }
}

console.log(`📊 Validation Results:`);
console.log(`   Total images: ${totalImages}`);
console.log(`   Missing images: ${missingImages.length}`);
console.log(`   Missing alt text: ${missingAltText.length}\n`);

let hasErrors = false;

if (missingImages.length > 0) {
  console.error('❌ Missing image files:\n');
  missingImages.forEach(img => {
    console.error(`   ${img.file}`);
    console.error(`   └─ ${img.image} (not found at ${img.resolved})\n`);
  });
  hasErrors = true;
}

if (missingAltText.length > 0) {
  console.error('❌ Images missing alt text:\n');
  missingAltText.forEach(img => {
    console.error(`   ${img.file}`);
    console.error(`   └─ ${img.image}\n`);
  });
  hasErrors = true;
}

if (hasErrors) {
  process.exit(1);
} else {
  console.log('✅ All images are valid!');
}
```

```javascript
// scripts/validate-all.js
import { execSync } from 'child_process';
import fs from 'fs';

const validations = [
  {
    name: 'Link Validation',
    command: 'npm run validate:links',
    critical: true
  },
  {
    name: 'Code Example Validation',
    command: 'npm run validate:examples',
    critical: true
  },
  {
    name: 'Markdown Linting',
    command: 'npm run lint:md',
    critical: false
  },
  {
    name: 'Image Reference Validation',
    command: 'npm run validate:images',
    critical: true
  },
  {
    name: 'Frontmatter Validation',
    command: 'npm run validate:frontmatter',
    critical: false
  }
];

console.log('🔍 Running comprehensive documentation validation...\n');
console.log('='.repeat(60) + '\n');

const results = [];
let criticalFailures = 0;

for (const validation of validations) {
  console.log(`\n${'='.repeat(60)}`);
  console.log(`Running: ${validation.name}`);
  console.log('='.repeat(60) + '\n');

  const startTime = Date.now();

  try {
    execSync(validation.command, {
      stdio: 'inherit',
      env: { ...process.env, FORCE_COLOR: '1' }
    });

    const duration = ((Date.now() - startTime) / 1000).toFixed(2);

    results.push({
      name: validation.name,
      status: 'PASS',
      critical: validation.critical,
      duration
    });

    console.log(`\n✅ ${validation.name} passed (${duration}s)\n`);
  } catch (error) {
    const duration = ((Date.now() - startTime) / 1000).toFixed(2);

    results.push({
      name: validation.name,
      status: 'FAIL',
      critical: validation.critical,
      duration
    });

    console.error(`\n❌ ${validation.name} failed (${duration}s)\n`);

    if (validation.critical) {
      criticalFailures++;
    }
  }
}

// Generate summary
console.log('\n' + '='.repeat(60));
console.log('📊 Validation Summary');
console.log('='.repeat(60) + '\n');

results.forEach(r => {
  const icon = r.status === 'PASS' ? '✅' : '❌';
  const critical = r.critical ? '[CRITICAL]' : '[WARNING]';
  console.log(`${icon} ${r.name.padEnd(35)} ${critical.padEnd(12)} ${r.duration}s`);
});

console.log('\n' + '='.repeat(60));

// Generate report file
const report = generateMarkdownReport(results);
fs.writeFileSync('validation-report.md', report);
console.log('📄 Report saved: validation-report.md\n');

// Exit with appropriate code
if (criticalFailures > 0) {
  console.error(`\n❌ ${criticalFailures} critical validation(s) failed.`);
  console.error('   Fix issues before deployment.\n');
  process.exit(1);
} else {
  console.log('\n✅ All validations passed!\n');
  process.exit(0);
}

function generateMarkdownReport(results) {
  const timestamp = new Date().toISOString();
  const passed = results.filter(r => r.status === 'PASS').length;
  const failed = results.filter(r => r.status === 'FAIL').length;

  return `# Documentation Validation Report

**Generated:** ${timestamp}

## Summary

- **Total Validations:** ${results.length}
- **Passed:** ${passed}
- **Failed:** ${failed}

## Results

| Validation | Status | Type | Duration |
|------------|--------|------|----------|
${results.map(r =>
  `| ${r.name} | ${r.status === 'PASS' ? '✅ PASS' : '❌ FAIL'} | ${r.critical ? 'Critical' : 'Warning'} | ${r.duration}s |`
).join('\n')}

## Next Steps

${failed > 0 ? `
### Fix Issues

${results.filter(r => r.status === 'FAIL').map(r => `
- **${r.name}**: Review logs above for details
`).join('')}
` : '✅ All validations passed. Ready for deployment!'}

---

*Generated by validate-all.js*
`;
}
```

### Common Issues and Solutions

**Issue 1: False positive external link failures**
- Solution: Add to .validation-config.json ignore list
- Or use manual override: `--ignore-external-links`
- Check if link temporarily down vs permanently broken

**Issue 2: Code example TypeScript errors**
- Solution: Add `// @ts-nocheck` to skip validation
- Or fix TypeScript code to compile
- Check imports are mocked correctly

**Issue 3: Internal links broken after VitePress migration**
- Solution: Update links to VitePress clean URL format
- Remove .md extensions
- Use absolute paths from site root

**Issue 4: Image paths not resolving**
- Solution: Move images to docs/public/
- Use absolute paths from public: `/images/logo.png`
- Or relative paths from markdown file

**Issue 5: Validation timeout in CI/CD**
- Solution: Increase external link timeout
- Reduce number of concurrent checks
- Skip known slow external links

**Issue 6: Frontmatter validation too strict**
- Solution: Update validation rules
- Make description optional for certain page types
- Document required frontmatter in CONTRIBUTING.md

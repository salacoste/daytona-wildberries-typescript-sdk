```typescript
export enum MCPErrorCode {
  // Standard JSON-RPC 2.0 errors
  PARSE_ERROR = -32700,
  INVALID_REQUEST = -32600,
  METHOD_NOT_FOUND = -32601,
  INVALID_PARAMS = -32602,
  INTERNAL_ERROR = -32603,

  // MCP-specific errors
  AUTHENTICATION_ERROR = -32001,
  RATE_LIMIT_ERROR = -32002,
  VALIDATION_ERROR = -32003,
  TIMEOUT_ERROR = -32004,
  RESOURCE_NOT_FOUND = -32005,
  SDK_ERROR = -32006
}

export interface MCPError {
  code: MCPErrorCode;
  message: string;
  data?: {
    detail?: string;
    context?: Record<string, any>;
    recoveryHint?: string;
  };
}
```

**Error Handler Implementation:**

```typescript
// mcp-server/src/errors/error-handler.ts
import { WBAPIError, AuthenticationError, RateLimitError, ValidationError, NetworkError } from 'daytona-wildberries-typescript-sdk';
import { MCPError, MCPErrorCode } from './mcp-error-codes';
import { logger } from '../logging/logger';

export class MCPErrorHandler {
  handleError(error: unknown, context: { tool?: string; args?: any }): MCPError {
    // SDK-specific errors
    if (error instanceof AuthenticationError) {
      logger.error({ error, context }, 'Authentication error');
      return {
        code: MCPErrorCode.AUTHENTICATION_ERROR,
        message: 'Authentication failed',
        data: {
          detail: error.message,
          recoveryHint: 'Check WB_API_KEY environment variable. Get API key from Wildberries seller portal.'
        }
      };
    }

    if (error instanceof RateLimitError) {
      logger.warn({ error, context, retryAfter: error.retryAfter }, 'Rate limit exceeded');
      return {
        code: MCPErrorCode.RATE_LIMIT_ERROR,
        message: 'Rate limit exceeded',
        data: {
          detail: error.message,
          context: { retryAfter: error.retryAfter },
          recoveryHint: `Wait ${error.retryAfter}ms before retrying. SDK will automatically retry.`
        }
      };
    }

    if (error instanceof ValidationError) {
      logger.warn({ error, context }, 'Validation error');
      return {
        code: MCPErrorCode.VALIDATION_ERROR,
        message: 'Invalid parameters',
        data: {
          detail: error.message,
          context: { fieldErrors: (error as any).fieldErrors },
          recoveryHint: 'Check parameter types and required fields in tool schema.'
        }
      };
    }

    if (error instanceof NetworkError) {
      logger.error({ error, context, isTimeout: (error as any).isTimeout }, 'Network error');
      return {
        code: MCPErrorCode.TIMEOUT_ERROR,
        message: 'Network request failed',
        data: {
          detail: error.message,
          recoveryHint: (error as any).isTimeout
            ? 'Request timed out. Try again or increase timeout with WB_SDK_TIMEOUT.'
            : 'Network connectivity issue. Check internet connection and Wildberries API status.'
        }
      };
    }

    // MCP protocol errors
    if (error instanceof SyntaxError) {
      logger.error({ error, context }, 'JSON parse error');
      return {
        code: MCPErrorCode.PARSE_ERROR,
        message: 'Invalid JSON in request',
        data: { detail: error.message }
      };
    }

    // Generic error
    logger.error({ error, context }, 'Unexpected error');
    return {
      code: MCPErrorCode.INTERNAL_ERROR,
      message: 'Internal server error',
      data: {
        detail: error instanceof Error ? error.message : 'Unknown error',
        recoveryHint: 'This is an unexpected error. Please report to MCP server maintainers.'
      }
    };
  }

  handleConnectionError(error: Error, attemptNumber: number): void {
    logger.error({ error, attemptNumber }, 'Connection error');

    if (attemptNumber < 3) {
      const delay = Math.pow(2, attemptNumber) * 1000; // 1s, 2s, 4s
      logger.info({ delay, attemptNumber }, 'Retrying connection');
      // Retry logic handled by caller
    } else {
      logger.error('Max connection retries reached. Server shutting down.');
      process.exit(1);
    }
  }
}
```

**Request Logger Middleware:**

```typescript
// mcp-server/src/logging/request-logger.ts
import { performance } from 'perf_hooks';
import { logger } from './logger';

export class RequestLogger {
  logToolInvocation(toolName: string, args: any, executionTime: number, status: 'success' | 'error', error?: Error) {
    const sanitizedArgs = this.sanitizeParameters(args);

    logger.info({
      type: 'tool_invocation',
      tool: toolName,
      args: sanitizedArgs,
      executionTime,
      status,
      ...(error && { error: error.message })
    }, `Tool ${toolName} ${status}`);
  }

  logPerformanceMetrics(toolName: string, metrics: {
    totalTime: number;
    sdkTime: number;
    mappingTime: number;
  }) {
    if (process.env.WB_MCP_DEBUG === 'true') {
      logger.debug({
        type: 'performance_metrics',
        tool: toolName,
        ...metrics,
        overhead: metrics.totalTime - metrics.sdkTime
      }, `Performance metrics for ${toolName}`);
    }
  }

  private sanitizeParameters(args: any): any {
    // Remove sensitive fields
    const sensitiveFields = ['apiKey', 'password', 'token', 'authorization'];

    const sanitize = (obj: any): any => {
      if (typeof obj !== 'object' || obj === null) return obj;

      const sanitized: any = Array.isArray(obj) ? [] : {};

      for (const [key, value] of Object.entries(obj)) {
        if (sensitiveFields.includes(key.toLowerCase())) {
          sanitized[key] = '[REDACTED]';
        } else if (typeof value === 'object') {
          sanitized[key] = sanitize(value);
        } else {
          sanitized[key] = value;
        }
      }

      return sanitized;
    };

    return sanitize(args);
  }
}
```

**Health Check Endpoint:**

```typescript
// mcp-server/src/health/health-check.ts
export interface HealthStatus {
  status: 'healthy' | 'degraded' | 'unhealthy';
  uptime: number;
  version: string;
  sdk: {
    connected: boolean;
    lastPing?: number;
  };
  stats: {
    totalRequests: number;
    successfulRequests: number;
    failedRequests: number;
    averageResponseTime: number;
  };
}

export class HealthChecker {
  private startTime = Date.now();
  private stats = {
    totalRequests: 0,
    successfulRequests: 0,
    failedRequests: 0,
    totalResponseTime: 0
  };

  async checkHealth(): Promise<HealthStatus> {
    const sdkConnected = await this.checkSDKConnection();

    const status = this.determineStatus(sdkConnected);

    return {
      status,
      uptime: Date.now() - this.startTime,
      version: process.env.npm_package_version || '1.0.0',
      sdk: {
        connected: sdkConnected,
        lastPing: this.lastSuccessfulPing
      },
      stats: {
        totalRequests: this.stats.totalRequests,
        successfulRequests: this.stats.successfulRequests,
        failedRequests: this.stats.failedRequests,
        averageResponseTime: this.stats.totalRequests > 0
          ? this.stats.totalResponseTime / this.stats.totalRequests
          : 0
      }
    };
  }

  private async checkSDKConnection(): Promise<boolean> {
    try {
      await this.sdk.general.ping();
      this.lastSuccessfulPing = Date.now();
      return true;
    } catch {
      return false;
    }
  }

  private determineStatus(sdkConnected: boolean): 'healthy' | 'degraded' | 'unhealthy' {
    const errorRate = this.stats.failedRequests / Math.max(this.stats.totalRequests, 1);

    if (!sdkConnected) return 'unhealthy';
    if (errorRate > 0.5) return 'degraded';
    return 'healthy';
  }

  recordRequest(success: boolean, responseTime: number) {
    this.stats.totalRequests++;
    if (success) {
      this.stats.successfulRequests++;
    } else {
      this.stats.failedRequests++;
    }
    this.stats.totalResponseTime += responseTime;
  }
}
```

**Graceful Shutdown:**

```typescript
// mcp-server/src/lifecycle/shutdown.ts
export class GracefulShutdown {
  private inFlightRequests = new Set<string>();
  private isShuttingDown = false;

  trackRequest(requestId: string) {
    this.inFlightRequests.add(requestId);
  }

  completeRequest(requestId: string) {
    this.inFlightRequests.delete(requestId);
  }

  async shutdown(): Promise<void> {
    if (this.isShuttingDown) return;

    this.isShuttingDown = true;
    logger.info({ inFlightRequests: this.inFlightRequests.size }, 'Starting graceful shutdown');

    // Stop accepting new requests
    this.server.stopAcceptingRequests();

    // Wait for in-flight requests (max 30 seconds)
    const maxWait = 30000;
    const checkInterval = 100;
    let waited = 0;

    while (this.inFlightRequests.size > 0 && waited < maxWait) {
      await new Promise(resolve => setTimeout(resolve, checkInterval));
      waited += checkInterval;
    }

    if (this.inFlightRequests.size > 0) {
      logger.warn({
        remainingRequests: this.inFlightRequests.size
      }, 'Shutdown timeout reached with pending requests');
    } else {
      logger.info('All requests completed. Shutting down cleanly.');
    }

    await this.server.close();
    process.exit(0);
  }

  setupSignalHandlers() {
    process.on('SIGTERM', () => this.shutdown());
    process.on('SIGINT', () => this.shutdown());
  }
}
```

```typescript
// mcp-server/tests/performance/load-test.ts
import { performance } from 'perf_hooks';

describe('Load Testing', () => {
  it('should handle 100+ tool calls within performance budget', async () => {
    const calls = 100;
    const maxAverageTime = 500; // ms

    const times: number[] = [];

    for (let i = 0; i < calls; i++) {
      const start = performance.now();
      await server.executeTool('products_list', { limit: 10 });
      times.push(performance.now() - start);
    }

    const average = times.reduce((a, b) => a + b, 0) / times.length;
    const p95 = times.sort((a, b) => a - b)[Math.floor(calls * 0.95)];

    expect(average).toBeLessThan(maxAverageTime);
    expect(p95).toBeLessThan(maxAverageTime * 2);

    console.log(`Average: ${average.toFixed(2)}ms, P95: ${p95.toFixed(2)}ms`);
  });
});
```

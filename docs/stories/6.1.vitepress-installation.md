### VitePress Configuration Template

**Location:** `.vitepress/config.ts`

**Basic Configuration (TypeScript):**
```typescript
import { defineConfig } from 'vitepress';

export default defineConfig({
  // Site metadata
  title: 'Wildberries TypeScript SDK',
  description: 'Full-featured SDK providing type-safe access to all Wildberries marketplace API methods',
  base: '/', // Adjust if using GitHub Pages with repo name: '/repo-name/'

  // Source directory
  srcDir: '../docs', // VitePress serves from docs/ directory

  // Output directory
  outDir: '.vitepress/dist',

  // Theme configuration
  themeConfig: {
    // Site title in navigation
    siteTitle: 'Wildberries SDK',

    // Navigation (placeholder, will expand in Story 6.4)
    nav: [
      { text: 'Home', link: '/' },
      { text: 'Guide', link: '/getting-started/quickstart' }
    ],

    // Sidebar (placeholder, will expand in Story 6.2)
    sidebar: [],

    // Social links
    socialLinks: [
      { icon: 'github', link: 'https://github.com/salacoste/daytona-wildberries-typescript-sdk' }
    ]
  }
});
```
